# #######################################################################################################################

find_package(DirectXTex CONFIG REQUIRED)
find_package(tbb CONFIG REQUIRED)
find_package(absl CONFIG REQUIRED)
find_package(re2 CONFIG REQUIRED)
find_package(glaze CONFIG REQUIRED)
find_package(lz4 CONFIG REQUIRED)

# #######################################################################################################################
# # Find dependencies - Git
# #######################################################################################################################
include(FetchContent)

# Boost Reflect
# https://github.com/qlibs/reflect
FetchContent_Declare(
    boost_ext_reflect
    GIT_REPOSITORY https://github.com/Arial6502/reflect/
    GIT_SHALLOW ON
    GIT_TAG ab2bad2e5600e0890380f0226412d91ffaed3c39
)
FetchContent_MakeAvailable(boost_ext_reflect)

target_include_directories(
	${boost_ext_reflect_SOURCE_DIR}
)

target_link_libraries(
	TBB::tbb
	absl::base
	absl::flat_hash_map
	re2::re2
	lz4::lz4
	Microsoft::DirectXTex
	glaze::glaze
)
